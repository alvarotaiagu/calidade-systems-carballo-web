# =============================================================================
#  Calidade Systems - .htaccess para convivir con el WordPress existente
#  Generado el 2026-09-17
#
#  Colocar esta web estatica EN LA MISMA carpeta raiz que WordPress, de modo
#  que existan de verdad /es/index.html, /en/index.html, /assets/, /css/, /js/.
#
#  Estas reglas van ANTES del bloque `# BEGIN WordPress`. Con eso basta:
#    - /es/ y /en/ son directorios reales con index.html -> mod_dir los sirve
#      y la peticion nunca llega al rewrite de WordPress.
#    - /es/2019/01/10/slug/ no existe en disco -> cae en el rewrite de
#      WordPress y se sirve el articulo con 200, igual que hoy.
#    - las URLs migradas se cortan aqui con un 301 antes de llegar a WP.
# =============================================================================

Options -MultiViews
DirectoryIndex index.html index.php

<IfModule mod_rewrite.c>
RewriteEngine On

# --- dominio canonico con www ---
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^(.*)$ https://www.calidade.systems/$1 [R=301,L]

# --- la raiz sirve espanol (hoy hace 302 a /en/) ---
RewriteRule ^$ /es/ [R=301,L]

  ExpiresActive On
  <FilesMatch "\.(svg|png|jpg|jpeg|webp|woff2)$">
    ExpiresDefault "access plus 30 days"
  </FilesMatch>
  <FilesMatch "\.(css|js)$">
